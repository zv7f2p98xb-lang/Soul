SG_FN float sg_fogf(int mode, float d, float start, float end, float dens) {
    float f = 1.0f;
    if (mode == 1) f = sg_expf(-d * dens);
    else if (mode == 2) f = sg_expf(-(d * dens) * (d * dens));
    else if (mode == 3) f = end != start ? (end - d) / (end - start) : (d < start ? 1.0f : 0.0f);
    return sg_sat1(f);
}
// ---- GL fixed-function texture environment (sgpu.c tex_combine) ----
SG_FN float4 gl_combine(int env, int fmt, float4 c, float4 tx, float4 ec) {
    int has_rgb = fmt != 2;
    int has_a = fmt == 0 || fmt == 2 || fmt == 4 || fmt == 5;
    if (env == 1) {
        if (has_rgb) { c.x = tx.x; c.y = tx.y; c.z = tx.z; }
        if (has_a) c.w = tx.w;
    } else if (env == 2) {
        if (fmt == 0) { c.x += (tx.x - c.x) * tx.w; c.y += (tx.y - c.y) * tx.w; c.z += (tx.z - c.z) * tx.w; }
        else if (fmt == 1) { c.x = tx.x; c.y = tx.y; c.z = tx.z; }
    } else if (env == 3) {
        if (has_rgb) { c.x += tx.x; c.y += tx.y; c.z += tx.z; if (c.x > 1.0f) c.x = 1.0f; if (c.y > 1.0f) c.y = 1.0f; if (c.z > 1.0f) c.z = 1.0f; }
        if (has_a) c.w *= tx.w;
    } else if (env == 4) {
        if (has_rgb) { c.x = c.x * (1.0f - tx.x) + ec.x * tx.x; c.y = c.y * (1.0f - tx.y) + ec.y * tx.y; c.z = c.z * (1.0f - tx.z) + ec.z * tx.z; }
        if (has_a) c.w *= tx.w;
    } else {
        if (has_rgb) { c.x *= tx.x; c.y *= tx.y; c.z *= tx.z; }
        if (has_a) c.w *= tx.w;
    }
    return c;
}

SG_FN float4 ff_argx(int arg, float4 tx, float4 cur, float4 tmp, float4 dif, float4 spc, float4 tf, float4 cst, int alpha_only) {
    float4 o;
    int k = arg & 15;
    if (k == 0) o = dif; else if (k == 1) o = cur; else if (k == 2) o = tx; else if (k == 3) o = tf;
    else if (k == 4) o = spc; else if (k == 5) o = tmp; else if (k == 6) o = cst; else o = cur;
    if (arg & 0x10) o = mk4(1.0f - o.x, 1.0f - o.y, 1.0f - o.z, 1.0f - o.w);
    if ((arg & 0x20) && !alpha_only) { float w = o.w; o = mk4(w, w, w, w); }
    return o;
}

SG_FN float ff_op1(int op, float x0, float x1, float x2, float a1w, float bdif, float btex, float btf, float bcur, int alpha) {
    float v = x1, bf;
    switch (op) {
    case 2: v = x1; break; case 3: v = x2; break;
    case 4: v = x1 * x2; break; case 5: v = x1 * x2 * 2.0f; break; case 6: v = x1 * x2 * 4.0f; break;
    case 7: v = x1 + x2; break; case 8: v = x1 + x2 - 0.5f; break; case 9: v = (x1 + x2 - 0.5f) * 2.0f; break;
    case 10: v = x1 - x2; break; case 11: v = x1 + x2 - x1 * x2; break;
    case 12: bf = bdif; v = x1 * bf + x2 * (1.0f - bf); break;
    case 13: bf = btex; v = x1 * bf + x2 * (1.0f - bf); break;
    case 14: bf = btf; v = x1 * bf + x2 * (1.0f - bf); break;
    case 15: v = x1 + x2 * (1.0f - btex); break;
    case 16: bf = bcur; v = x1 * bf + x2 * (1.0f - bf); break;
    case 17: v = x1; break;
    case 18: v = alpha ? x1 : x1 + a1w * x2; break;
    case 19: v = alpha ? x1 : x1 * x2 + a1w; break;
    case 20: v = alpha ? x1 : (1.0f - a1w) * x2 + x1; break;
    case 21: v = alpha ? x1 : (1.0f - x1) * x2 + a1w; break;
    case 25: v = x0 + x1 * x2; break;
    case 26: v = x0 * x1 + (1.0f - x0) * x2; break;
    default: v = x1; break;
    }
    return sg_sat1(v);
}
SG_FN float4 ff_opc(int op, float4 a0, float4 a1, float4 a2, float4 tx, float4 cur, float4 dif, float4 tf) {
    return mk4(ff_op1(op, a0.x, a1.x, a2.x, a1.w, dif.w, tx.w, tf.w, cur.w, 0), ff_op1(op, a0.y, a1.y, a2.y, a1.w, dif.w, tx.w, tf.w, cur.w, 0),
               ff_op1(op, a0.z, a1.z, a2.z, a1.w, dif.w, tx.w, tf.w, cur.w, 0), 0.0f);
}
SG_FN float ff_opa(int op, float4 a0, float4 a1, float4 a2, float4 tx, float4 cur, float4 dif, float4 tf) {
    return ff_op1(op, a0.w, a1.w, a2.w, a1.w, dif.w, tx.w, tf.w, cur.w, 1);
}

SG_FN float4 ff_dot3(float4 a1, float4 a2) {
    float d = (a1.x * 2.0f - 1.0f) * (a2.x * 2.0f - 1.0f) + (a1.y * 2.0f - 1.0f) * (a2.y * 2.0f - 1.0f) + (a1.z * 2.0f - 1.0f) * (a2.z * 2.0f - 1.0f);
    return sg_splat(sg_sat1(d));
}

SG_FN float4 d3_mod(int mod, float4 v) {
    float4 r = v;
    switch (mod) {
    case 1: r = mk4(-v.x, -v.y, -v.z, -v.w); break;
    case 2: r = mk4(v.x - 0.5f, v.y - 0.5f, v.z - 0.5f, v.w - 0.5f); break;
    case 3: r = mk4(-(v.x - 0.5f), -(v.y - 0.5f), -(v.z - 0.5f), -(v.w - 0.5f)); break;
    case 4: r = mk4(2.0f * (v.x - 0.5f), 2.0f * (v.y - 0.5f), 2.0f * (v.z - 0.5f), 2.0f * (v.w - 0.5f)); break;
    case 5: r = mk4(-2.0f * (v.x - 0.5f), -2.0f * (v.y - 0.5f), -2.0f * (v.z - 0.5f), -2.0f * (v.w - 0.5f)); break;
    case 6: r = mk4(1.0f - v.x, 1.0f - v.y, 1.0f - v.z, 1.0f - v.w); break;
    case 7: r = mk4(v.x * 2.0f, v.y * 2.0f, v.z * 2.0f, v.w * 2.0f); break;
    case 8: r = mk4(v.x * -2.0f, v.y * -2.0f, v.z * -2.0f, v.w * -2.0f); break;
    case 9: { float z = v.z; r = mk4(z != 0.0f ? v.x / z : v.x, z != 0.0f ? v.y / z : v.y, z != 0.0f ? v.z / z : v.z, z != 0.0f ? v.w / z : v.w); break; }
    case 10: { float w = v.w; r = mk4(w != 0.0f ? v.x / w : v.x, w != 0.0f ? v.y / w : v.y, w != 0.0f ? v.z / w : v.z, w != 0.0f ? v.w / w : v.w); break; }
    case 11: r = mk4(sg_fabs(v.x), sg_fabs(v.y), sg_fabs(v.z), sg_fabs(v.w)); break;
    case 12: r = mk4(-sg_fabs(v.x), -sg_fabs(v.y), -sg_fabs(v.z), -sg_fabs(v.w)); break;
    default: break;
    }
    return r;
}

SG_FN float4 d_absd(float4 raw, float4 d, int neg) {
    float sx = raw.x < 0.0f ? -1.0f : 1.0f, sy = raw.y < 0.0f ? -1.0f : 1.0f, sz = raw.z < 0.0f ? -1.0f : 1.0f, sw = raw.w < 0.0f ? -1.0f : 1.0f;

    float v = 0.0f; v += x.x * b.x + a.x * y.x; v += x.y * b.y + a.y * y.y; v += x.z * b.z + a.z * y.z; v += x.w * b.w + a.w * y.w; return sg_splat(v);
}
SG_FN float4 d_rcp(float4 r, float4 x) { return mk4(-r.x * r.x * x.x, -r.y * r.y * x.x, -r.z * r.z * x.x, -r.w * r.w * x.x); }
SG_FN float4 d_rsq(float4 r, float4 x) { return mk4(-0.5f * r.x * r.x * r.x * x.x, -0.5f * r.y * r.y * r.y * x.x, -0.5f * r.z * r.z * r.z * x.x, -0.5f * r.w * r.w * r.w * x.x); }
SG_FN float d_min1(float a, float b, float x, float y) { return a < b ? x : y; }
SG_FN float d_max1(float a, float b, float x, float y) { return a >= b ? x : y; }
SG_FN float4 d_min(float4 a, float4 b, float4 x, float4 y) { return mk4(d_min1(a.x, b.x, x.x, y.x), d_min1(a.y, b.y, x.y, y.y), d_min1(a.z, b.z, x.z, y.z), d_min1(a.w, b.w, x.w, y.w)); }
SG_FN float4 d_max(float4 a, float4 b, float4 x, float4 y) { return mk4(d_max1(a.x, b.x, x.x, y.x), d_max1(a.y, b.y, x.y, y.y), d_max1(a.z, b.z, x.z, y.z), d_max1(a.w, b.w, x.w, y.w)); }
SG_FN float d_cmp1(float a, float y, float z) { return a >= 0.0f ? y : z; }
SG_FN float d_cnd1(float a, float y, float z) { return a > 0.5f ? y : z; }
SG_M3(d_cmp, d_cmp1)
SG_M3(d_cnd, d_cnd1)
SG_FN float4 d_pow(float4 a, float4 b, float4 res, float4 x, float4 y) {
    float xa = sg_fabs(a.x), r = res.x, v = 0.0f;
    if (xa > 0.0f && sg_isfinite(r)) v = b.x * r / xa * (a.x < 0.0f ? -x.x : x.x) + r * sg_logf(xa) * y.x;
    return sg_splat(v);
}
SG_FN float4 d_exp(float4 r, float4 x) { return mk4(0.69314718f * r.x * x.x, 0.69314718f * r.y * x.x, 0.69314718f * r.z * x.x, 0.69314718f * r.w * x.x); }
SG_FN float4 d_log(float4 a, float4 x) { return sg_splat(a.x != 0.0f ? x.x / (sg_fabs(a.x) * 0.69314718f) * (a.x < 0.0f ? -1.0f : 1.0f) : 0.0f); }
SG_FN float4 d_sincos(float4 a, float4 x) { return mk4(-sg_sin(a.x) * x.x, sg_cos(a.x) * x.x, 0.0f, 0.0f); }
